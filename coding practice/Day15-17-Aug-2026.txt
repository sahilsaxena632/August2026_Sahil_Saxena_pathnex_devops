# Created a practice folder for day 15 pathnex tasks
mkdir pathnex-day15
cd pathnex-day15


# Wrote ansible playbook to deploy web app to multiple hosts
cat > deploy-web.yml << 'EOF'
- name: Deploy web application to multiple hosts
  hosts: all
  become: yes
  tasks:
    - name: Copy web app files
      copy:
        src: /path/to/web/app/
        dest: /var/www/html/
    - name: Start web app service
      service:
        name: apache2
        state: started
        enabled: yes
EOF
ansible-playbook deploy-web.yml


# Wrote terraform ec2 with alb
cat > alb-ec2.tf << 'EOF'
resource "aws_lb" "pathnex_lb" {
  name               = "pathnex-lb"
  internal           = false
  load_balancer_type = "application"
  security_groups    = [aws_security_group.allow_ssh_http.id]
  subnets            = [aws_subnet.public.id]
}

resource "aws_instance" "pathnex_ec2" {
  ami             = "ami-0abcd1234abcd1234"
  instance_type   = "t2.micro"
  security_groups = [aws_security_group.allow_ssh_http.name]
  tags = {
    Name = "Pathnex-EC2"
  }
}
EOF
terraform init
terraform plan

# Wrote Jenkinsfile multi environment deploy
cat > Jenkinsfile << 'EOF'
pipeline {
    agent any
    environment {
        IMAGE_NAME = 'pathnex-web-app'
        DEPLOY_ENV = 'production'
    }
    stages {
        stage('Build') {
            steps {
                echo 'Building Docker image...'
                script {
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }
        stage('Deploy to Kubernetes') {
            steps {
                script {
                    if (env.DEPLOY_ENV == 'production') {
                        sh 'kubectl apply -f kubernetes/prod-deployment.yaml'
                    } else {
                        sh 'kubectl apply -f kubernetes/dev-deployment.yaml'
                    }
                }
            }
        }
    }
}
EOF
cat Jenkinsfile


# Wrote GitLab CI with env specific deploy
cat > .gitlab-ci.yml << 'EOF'
stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t pathnex-web-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push pathnex-web-app

deploy:
  stage: deploy
  script:
    - |
      if [ "$CI_COMMIT_REF_NAME" == "main" ]; then
        kubectl apply -f kubernetes/prod-deployment.yaml
      else
        kubectl apply -f kubernetes/dev-deployment.yaml
      fi
EOF
cat .gitlab-ci.yml
