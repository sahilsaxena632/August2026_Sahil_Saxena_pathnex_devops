mkdir pathnex-day7

# Wrote ansible playbook for nginx with custom config
cat > nginx.yml << 'EOF'
- name: Setup Nginx with Custom Config
  hosts: all
  become: yes
  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present
    - name: Copy custom nginx.conf
      copy:
        src: nginx.conf
        dest: /etc/nginx/nginx.conf
    - name: Start nginx service
      service:
        name: nginx
        state: started
        enabled: yes
EOF
ansible-playbook nginx.yml


# Wrote terraform for ec2 with iam role
cat > iam-ec2.tf << 'EOF'
resource "aws_iam_role" "ec2_role" {
  name               = "pathnex_ec2_role"
  assume_role_policy = jsonencode({
    Version = "2012-10-17"
    Statement = [
      {
        Action = "sts:AssumeRole"
        Effect = "Allow"
        Principal = {
          Service = "ec2.amazonaws.com"
        }
      },
    ]
  })
}

resource "aws_instance" "pathnex_ec2" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "t2.medium"
  iam_instance_profile = aws_iam_role.ec2_role.name
  tags = {
    Name = "Pathnex-EC2"
  }
}
EOF
terraform init
terraform plan

# Built a java docker app
cat > Hello.java << 'EOF'
class Hello {
    public static void main(String[] args) {
        System.out.println("Hello Pathnex");
    }
}
EOF
cat > Dockerfile << 'EOF'
FROM openjdk:17
WORKDIR /opt/pathnex/java-app
COPY Hello.java /opt/pathnex/java-app/
RUN javac /opt/pathnex/java-app/Hello.java
CMD ["java", "-cp", "/opt/pathnex/java-app", "Hello"]
EOF
docker build -t pathnex-java .
docker run pathnex-java
