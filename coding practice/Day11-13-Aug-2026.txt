# Connected to the EC2 instance using SSH and with key pair i created
# Created a practice folder for day 11 pathnex tasks
mkdir pathnex-day11
cd pathnex-day11


# Wrote ansible playbook to install helm
cat > helm.yml << 'EOF'
- name: Install Helm
  hosts: all
  become: yes
  tasks:
    - name: Download Helm
      get_url:
        url: https://get.helm.sh/helm-v3.7.0-linux-amd64.tar.gz
        dest: /tmp/helm.tar.gz
    - name: Extract Helm
      unarchive:
        src: /tmp/helm.tar.gz
        dest: /usr/local/bin/
        remote_src: yes
    - name: Set Helm binary permissions
      file:
        path: /usr/local/bin/helm
        mode: '0755'
EOF
ansible-playbook helm.yml


# Wrote terraform for ec2 with elastic ip
cat > eip.tf << 'EOF'
resource "aws_eip" "pathnex_eip" {
  instance = aws_instance.pathnex_ec2.id
}

resource "aws_instance" "pathnex_ec2" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "t3.medium"
  tags = {
    Name = "Pathnex-EC2"
  }
}
EOF
terraform init
terraform plan


# Practiced helm install for nginx
helm repo add stable https://charts.helm.sh/stable
helm install pathnex-nginx stable/nginx-ingress
helm list


# Wrote Jenkinsfile multi stage checkout build push deploy
cat > Jenkinsfile << 'EOF'
pipeline {
    agent any
    environment {
        IMAGE_NAME = 'pathnex-web-app'
    }
    stages {
        stage('Checkout') {
            steps {
                git 'https://github.com/pathnex/sample-repo.git'
            }
        }
        stage('Build Docker Image') {
            steps {
                script {
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }
        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image("${env.IMAGE_NAME}").push('latest')
                    }
                }
            }
        }
        stage('Deploy to Kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}
EOF
cat Jenkinsfile


# Practiced dockerfile expose and port mapping
cat > Dockerfile << 'EOF'

EXPOSE 80
EOF
docker build -t pathnex-expose .
docker run -d -p 8080:80 pathnex-expose
