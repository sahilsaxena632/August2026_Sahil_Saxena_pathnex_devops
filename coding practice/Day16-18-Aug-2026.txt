# Connected to the EC2 instance using SSH and with key pair i created
# Created a practice folder for day 16 pathnex tasks
mkdir pathnex-day16
cd pathnex-day16


# Wrote ansible playbook to launch ec2 in two regions
cat > multi-region-ec2.yml << 'EOF'
- name: Create EC2 instances
  hosts: localhost
  become: yes
  tasks:
    - name: Launch EC2 instance in us-east-1
      ec2_instance:
        name: pathnex-ec2-us-east-1
        key_name: pathnex-key
        region: us-east-1
        instance_type: t2.micro
        image: ami-0abcd1234abcd1234
        wait: yes
    - name: Launch EC2 instance in us-west-1
      ec2_instance:
        name: pathnex-ec2-us-west-1
        key_name: pathnex-key
        region: us-west-1
        instance_type: t2.micro
        image: ami-0abcd1234abcd1234
        wait: yes
EOF
ansible-playbook multi-region-ec2.yml


# Wrote terraform for multi region vpc and ec2
cat > multi-region.tf << 'EOF'
resource "aws_vpc" "us_east_vpc" {
  cidr_block = "10.0.0.0/16"
}

resource "aws_vpc" "us_west_vpc" {
  cidr_block = "10.1.0.0/16"
}

resource "aws_instance" "pathnex_ec2_us_east" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "t2.micro"
}

resource "aws_instance" "pathnex_ec2_us_west" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "t2.micro"
}
EOF
terraform init
terraform plan


# Wrote kubernetes deployment and service
cat > deploy-svc.yaml << 'EOF'
apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-deployment
  labels:
    app: pathnex-app
spec:
  replicas: 2
  selector:
    matchLabels:
      app: pathnex-app
  template:
    metadata:
      labels:
        app: pathnex-app
    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
            - containerPort: 80
---
apiVersion: v1
kind: Service
metadata:
  name: pathnex-service
spec:
  selector:
    app: pathnex-app
  ports:
    - protocol: TCP
      port: 80
EOF
kubectl apply -f deploy-svc.yaml
kubectl get deploy
kubectl get svc


# Wrote Jenkinsfile to deploy using kubectl context
cat > Jenkinsfile << 'EOF'
pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building Docker image...'
                script {
                    docker.build('pathnex-web-app')
                }
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }
        stage('Deploy to Kubernetes') {
            steps {
                script {
                    if (env.CLUSTER == 'us-east') {
                        sh 'kubectl --context=us-east apply -f kubernetes/deployment.yaml'
                    } else {
                        sh 'kubectl --context=us-west apply -f kubernetes/deployment.yaml'
                    }
                }
            }
        }
    }
}
EOF
cat Jenkinsfile


# Built a docker image that runs a shell script
cat > start.sh << 'EOF'
#!/bin/bash
echo "Hello Pathnex"
EOF
chmod +x start.sh
cat > Dockerfile << 'EOF'
FROM ubuntu
WORKDIR /opt/pathnex/scripts
COPY start.sh /opt/pathnex/scripts/
RUN chmod +x /opt/pathnex/scripts/start.sh
CMD ["/opt/pathnex/scripts/start.sh"]
EOF
docker build -t pathnex-script .
docker run pathnex-script


# Removed practice files at the end of the session
cd ~
rm -rf pathnex-day16

terminated the ec2 instance
