# Connected to the EC2 instance using SSH and with key pair i created
# Created a practice folder for day 5 pathnex tasks
mkdir pathnex-day5
cd pathnex-day5


# Wrote a Jenkinsfile with git clone build test deploy
cat > Jenkinsfile << 'EOF'
pipeline {
    agent any
    stages {
        stage('Clone Repository') {
            steps {
                git 'https://github.com/pathnex/sample-repo.git'
            }
        }
        stage('Build') {
            steps {
                echo 'Building project from Git repository'
            }
        }
        stage('Test') {
            steps {
                echo 'Running unit tests'
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying application'
            }
        }
    }
}
EOF
cat Jenkinsfile


# Wrote GitLab CI with docker build push and kubectl apply
cat > .gitlab-ci.yml << 'EOF'
stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t pathnex-app .

push:
  stage: push
  script:
    - docker push pathnex-app

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml
EOF
cat .gitlab-ci.yml


# Wrote ansible playbook to create user pathnex
cat > user.yml << 'EOF'
- name: Create a new user
  hosts: all
  become: yes
  tasks:
    - name: Create user
      user:
        name: pathnex
        state: present
        shell: /bin/bash
    - name: Set permissions for the user
      file:
        path: /home/pathnex
        owner: pathnex
        group: pathnex
        mode: '0755'
EOF
ansible-playbook user.yml


# Wrote terraform for EC2 with tags and extra ebs volume
cat > main.tf << 'EOF'
resource "aws_instance" "PathnexEC2" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "t2.medium"
  tags = {
    Name = "Pathnex-Server"
  }
  ebs_block_device {
    device_name = "/dev/sdh"
    volume_size = 50
  }
}
EOF
terraform init
terraform plan


# Built a python docker app
mkdir -p python-app
echo 'print("Hello Pathnex")' > python-app/app.py
cat > Dockerfile << 'EOF'
FROM python:3.11
WORKDIR /opt/pathnex/python-app
COPY app.py /opt/pathnex/python-app/
CMD ["python", "/opt/pathnex/python-app/app.py"]
EOF
cp python-app/app.py .
docker build -t pathnex-python .
docker run pathnex-python


# Removed practice files at the end of the session
cd ~
rm -rf pathnex-day5

terminated the ec2 instance
