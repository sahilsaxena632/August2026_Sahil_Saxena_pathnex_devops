# Connected to the EC2 instance using SSH and with key pair i created
# Created a practice folder for day 8 pathnex tasks
mkdir pathnex-day8
cd pathnex-day8


# Wrote ansible playbook to install docker compose
cat > compose.yml << 'EOF'
- name: Install Docker Compose
  hosts: all
  become: yes
  tasks:
    - name: Install dependencies
      yum:
        name: "python3-pip"
        state: present
    - name: Install docker-compose
      pip:
        name: docker-compose
EOF
ansible-playbook compose.yml


# Wrote terraform ec2 with security group and ebs
cat > main.tf << 'EOF'
resource "aws_security_group" "allow_ssh_http" {
  name        = "allow_ssh_http"
  description = "Allow SSH and HTTP access"
  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }
  ingress {
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

resource "aws_instance" "pathnex_ec2" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "t3.medium"
  security_groups = [aws_security_group.allow_ssh_http.name]
  ebs_block_device {
    device_name = "/dev/sdh"
    volume_size = 30
  }
}
EOF
terraform init
terraform plan


# Wrote kubernetes NodePort service
cat > service.yaml << 'EOF'
apiVersion: v1
kind: Service
metadata:
  name: pathnex-service
spec:
  type: NodePort
  selector:
    app: pathnex-app
  ports:
    - protocol: TCP
      port: 80
      nodePort: 30007
EOF
kubectl apply -f service.yaml
kubectl get svc


# Wrote Jenkinsfile to build and push docker image
cat > Jenkinsfile << 'EOF'
pipeline {
    agent any
    stages {
        stage('Build Docker Image') {
            steps {
                script {
                    docker.build('pathnex-app')
                }
            }
        }
        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image('pathnex-app').push('latest')
                    }
                }
            }
        }
    }
}
EOF
cat Jenkinsfile


# Built a small alpine docker image
cat > Dockerfile << 'EOF'
FROM alpine:latest
WORKDIR /opt/pathnex/alpine-app
CMD ["echo", "Hello Pathnex"]
EOF
docker build -t pathnex-alpine .
docker run pathnex-alpine


# Removed practice files at the end of the session
cd ~
rm -rf pathnex-day8

terminated the ec2 instance
